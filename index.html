구글 스프레드시트(Apps Script) 연동 설정

💡 스프레드시트 연동 준비가 완료되었습니다!

구글 스프레드시트의 [확장 프로그램] → [Apps Script]를 웹 앱(Web App)으로 배포한 후, 발급받은 URL을 아래에 등록하면 원클릭으로 학생 명단을 불러오거나 배정 결과를 자동 기록할 수 있습니다.

URL을 입력하지 않더라도 브라우저 로컬 저장소(LocalStorage)에 실시간으로 안전하게 자동 보관됩니다.

📋 구글 시트에 붙여넣을 Apps Script (코드 보기 및 복사) 
// [구글 스프레드시트] - [확장 프로그램] - [Apps Script]에 붙여넣으세요.
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  // 1행 헤더(이름, 역할) 제외
  var students = [];
  var roles = [];
  for (var i = 1; i < data.length; i++) {
    if (data[i][0]) students.push(data[i][0]);
    if (data[i][1]) roles.push(data[i][1]);
  }
  return ContentService.createTextOutput(JSON.stringify({
    status: 'success',
    students: students,
    roles: roles
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var contents = JSON.parse(e.postData.contents);
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  sheet.clear();
  sheet.appendRow(["번호", "학생 이름", "배정 역할", "실천 임무", "배정 일시"]);
  
  contents.assignments.forEach(function(item) {
    sheet.appendRow([item.number, item.studentName, item.roleName, item.roleDesc, item.assignedAt]);
  });
  
  return ContentService.createTextOutput(JSON.stringify({
    status: 'success',
    message: '성공적으로 저장되었습니다.'
  })).setMimeType(ContentService.MimeType.JSON);
}
